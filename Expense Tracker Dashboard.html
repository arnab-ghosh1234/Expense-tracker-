import json import os from datetime import datetime DATA_FILE = "expenses_data.json" def load_data(): if os.path.exists(DATA_FILE): try: with open(DATA_FILE, "r", encoding="utf-8") as file: return json.load(file) except json.JSONDecodeError: return [] return [] def save_data(transactions): with open(DATA_FILE, "w", encoding="utf-8") as file: json.dump(transactions, file, indent=4, ensure_ascii=False) def add_transaction(transactions, trans_type): print("\n" + "-"*35) print("Add Income" if trans_type == "income" else "Add Expense") print("-"*35) try: amount_input = input("Amount (₹): ").strip() amount = float(amount_input) if amount <= 0: print("\nAmount cannot be zero or negative!") return category = input("Category (e.g., Food, Rent, Salary): ").strip() if not category: category = "General" description = input("Description/Note (Optional): ").strip() date = datetime.now().strftime("%Y-%m-%d %H:%M") transaction = { "id": len(transactions) + 1, "type": trans_type, "amount": amount, "category": category, "description": description if description else "No note", "date": date } transactions.append(transaction) save_data(transactions) print(f"\nSuccessfully added {'income' if trans_type == 'income' else 'expense'}!") except ValueError: print("\nInvalid input! Please enter a valid number for amount.") def view_summary(transactions): total_income = sum(t["amount"] for t in transactions if t["type"] == "income") total_expense = sum(t["amount"] for t in transactions if t["type"] == "expense") balance = total_income - total_expense print("\n" + "="*45) print(" SUMMARY") print("="*45) print(f"Total Income : ₹{total_income:,.2f}") print(f"Total Expense : ₹{total_expense:,.2f}") print("-" * 45) print(f"Current Balance: ₹{balance:,.2f}") print("="*45) def view_all_transactions(transactions): if not transactions: print("\nNo transactions recorded yet!") return print("\n" + "="*85) print(" ALL TRANSACTIONS") print("="*85) print(f"{'ID':<4} | {'Date & Time':<16} | {'Type':<8} | {'Category':<12} | {'Amount (₹)':<12} | {'Note':<15}") print("-" * 85) for t in transactions: type_str = "Income" if t["type"] == "income" else "Expense" print(f"{t['id']:<4} | {t['date']:<16} | {type_str:<8} | {t['category']:<12} | ₹{t['amount']:<11,.2f} | {t['description']:<15}") print("="*85) def main(): transactions = load_data() while True: print("\n" + "*"*40) print(" EXPENSE TRACKER") print("*"*40) print(" 1. Add Income") print(" 2. Add Expense") print(" 3. View All Transactions") print(" 4. View Summary") print(" 5. Exit") print("*"*40) choice = input("Enter your choice (1-5): ").strip() if choice == "1": add_transaction(transactions, "income") elif choice == "2": add_transaction(transactions, "expense") elif choice == "3": view_all_transactions(transactions) elif choice == "4": view_summary(transactions) elif choice == "5": print("\nExiting program. Goodbye!") break else: print("\nInvalid choice! Please select between 1 and 5.") if __name__ == "__main__": main()
